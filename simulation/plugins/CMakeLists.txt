cmake_minimum_required(VERSION 3.16)
project(mupt_sensor_follower LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(mupt_sensor_follower SHARED SensorFollower.cc)
target_compile_features(mupt_sensor_follower PRIVATE cxx_std_17)
target_link_libraries(mupt_sensor_follower PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
add_library(mupt_city_obstacles SHARED CityObstacles.cc)
target_compile_features(mupt_city_obstacles PRIVATE cxx_std_17)
target_link_libraries(mupt_city_obstacles PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
add_library(mupt_package_follower SHARED PackageFollower.cc)
target_compile_features(mupt_package_follower PRIVATE cxx_std_17)
target_link_libraries(mupt_package_follower PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
